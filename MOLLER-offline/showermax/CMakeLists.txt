cmake_minimum_required(VERSION 3.16)
project(MOLLERShowerMaxReplay VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(BUILD_REPLAY "Build the ROOT/Podd MOLLER replay library" ON)
if(BUILD_TESTING)
 add_executable(test_sm_sparse tests/test_sparse.cpp)
 target_include_directories(test_sm_sparse PRIVATE include)
 add_test(NAME sm_sparse COMMAND test_sm_sparse)
endif()
if(NOT BUILD_REPLAY)
 return()
endif()
find_package(ROOT REQUIRED COMPONENTS Core RIO Tree Hist Physics)
include(${ROOT_USE_FILE})
set(PODD_PREFIX "$ENV{ANALYZER}" CACHE PATH "Hall A Analyzer install prefix")
if(NOT PODD_PREFIX)
 find_program(ANALYZER_EXE analyzer)
 if(ANALYZER_EXE)
  get_filename_component(_bin "${ANALYZER_EXE}" DIRECTORY)
  get_filename_component(PODD_PREFIX "${_bin}" DIRECTORY)
 endif()
endif()
find_path(PODD_INCLUDE_DIR THaAnalyzer.h HINTS "${PODD_PREFIX}/include")
find_library(PODD_LIBRARY NAMES HallA Podd HINTS "${PODD_PREFIX}/lib64" "${PODD_PREFIX}/lib")
if(NOT PODD_INCLUDE_DIR OR NOT PODD_LIBRARY)
 message(FATAL_ERROR "Podd not found. Set -DPODD_PREFIX=/path/to/analyzer/install")
endif()
set(PODD_LIBRARIES ${PODD_LIBRARY})
foreach(_lib Podd dc)
 find_library(PODD_${_lib}_LIBRARY NAMES ${_lib} HINTS "${PODD_PREFIX}/lib64" "${PODD_PREFIX}/lib")
 if(PODD_${_lib}_LIBRARY)
  list(APPEND PODD_LIBRARIES ${PODD_${_lib}_LIBRARY})
 endif()
endforeach()
set(MD_PULSE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../include" CACHE PATH "Existing shared MDPulse.h directory")
if(NOT EXISTS "${MD_PULSE_INCLUDE_DIR}/MDPulse.h")
 message(FATAL_ERROR "Install this add-on under MOLLER-offline/showermax beside the timing-update include directory")
endif()
file(READ "${MD_PULSE_INCLUDE_DIR}/MDPulse.h" _pulse)
if(NOT _pulse MATCHES "half_height_time_ns")
 message(FATAL_ERROR "Apply the working MD timing update before building Shower-Max replay")
endif()
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include ${MD_PULSE_INCLUDE_DIR} ${PODD_INCLUDE_DIR})
ROOT_GENERATE_DICTIONARY(G__mollershowermax
 ${CMAKE_CURRENT_SOURCE_DIR}/include/MOLLERShowerMaxPodd.h
 ${CMAKE_CURRENT_SOURCE_DIR}/include/MOLLERShowerMaxDetector.h
 ${CMAKE_CURRENT_SOURCE_DIR}/include/MOLLERShowerMaxApparatus.h
 MODULE mollershowermax LINKDEF ${CMAKE_CURRENT_SOURCE_DIR}/include/ShowerMaxLinkDef.h)
add_library(mollershowermax SHARED src/MOLLERShowerMaxPodd.cxx src/MOLLERShowerMaxDetector.cxx src/MOLLERShowerMaxApparatus.cxx
 ${CMAKE_CURRENT_BINARY_DIR}/G__mollershowermax.cxx)
target_include_directories(mollershowermax PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
 $<INSTALL_INTERFACE:include> ${MD_PULSE_INCLUDE_DIR} ${PODD_INCLUDE_DIR})
target_link_libraries(mollershowermax PUBLIC ${PODD_LIBRARIES} ROOT::Core ROOT::RIO ROOT::Tree ROOT::Hist ROOT::Physics)
get_filename_component(PODD_LIBRARY_DIR ${PODD_LIBRARY} DIRECTORY)
set_target_properties(mollershowermax PROPERTIES INSTALL_RPATH "${PODD_LIBRARY_DIR};${ROOT_LIBRARY_DIR}")
install(TARGETS mollershowermax LIBRARY DESTINATION lib64)
install(DIRECTORY include/ DESTINATION include FILES_MATCHING PATTERN "*.h")
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/libmollershowermax_rdict.pcm
 ${CMAKE_CURRENT_BINARY_DIR}/libmollershowermax.rootmap DESTINATION lib64 OPTIONAL)
message(STATUS "Podd include: ${PODD_INCLUDE_DIR}")
message(STATUS "Podd libraries: ${PODD_LIBRARIES}")
