cmake_minimum_required(VERSION 3.16)
project(MOLLERTriggerScintillatorReplay VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(BUILD_REPLAY "Build the ROOT/Podd MOLLER replay library" ON)
set(MD_PULSE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../include" CACHE PATH "Existing shared MDPulse.h directory")
if(BUILD_TESTING)
 add_executable(test_ts_input tests/test_input.cpp)
 target_include_directories(test_ts_input PRIVATE include)
 add_test(NAME ts_input COMMAND test_ts_input "${CMAKE_CURRENT_SOURCE_DIR}/../../MOLLER-replay/DB_trigscint/MOLLER_trigscint_electronics_map.csv")
 add_executable(test_ts_pulse tests/test_pulse.cpp)
 target_include_directories(test_ts_pulse PRIVATE "${MD_PULSE_INCLUDE_DIR}")
 add_test(NAME ts_pulse COMMAND test_ts_pulse)
endif()
if(NOT BUILD_REPLAY)
 return()
endif()
find_package(ROOT REQUIRED COMPONENTS Core RIO Tree Hist Physics)
include(${ROOT_USE_FILE})
set(PODD_PREFIX "$ENV{ANALYZER}" CACHE PATH "Hall A Analyzer install prefix")
if(NOT PODD_PREFIX)
 find_program(ANALYZER_EXE analyzer)
 if(ANALYZER_EXE)
  get_filename_component(_bin "${ANALYZER_EXE}" DIRECTORY)
  get_filename_component(PODD_PREFIX "${_bin}" DIRECTORY)
 endif()
endif()
find_path(PODD_INCLUDE_DIR THaAnalyzer.h HINTS "${PODD_PREFIX}/include")
find_library(PODD_LIBRARY NAMES HallA Podd HINTS "${PODD_PREFIX}/lib64" "${PODD_PREFIX}/lib")
if(NOT PODD_INCLUDE_DIR OR NOT PODD_LIBRARY)
 message(FATAL_ERROR "Podd not found. Set -DPODD_PREFIX=/path/to/analyzer/install")
endif()
set(PODD_LIBRARIES ${PODD_LIBRARY})
foreach(_lib Podd dc)
 find_library(PODD_${_lib}_LIBRARY NAMES ${_lib} HINTS "${PODD_PREFIX}/lib64" "${PODD_PREFIX}/lib")
 if(PODD_${_lib}_LIBRARY)
  list(APPEND PODD_LIBRARIES ${PODD_${_lib}_LIBRARY})
 endif()
endforeach()
if(NOT EXISTS "${MD_PULSE_INCLUDE_DIR}/MDPulse.h")
 message(FATAL_ERROR "Install this add-on under MOLLER-offline/trigscint beside the timing-update include directory")
endif()
file(READ "${MD_PULSE_INCLUDE_DIR}/MDPulse.h" _pulse)
if(NOT _pulse MATCHES "half_height_time_ns")
 message(FATAL_ERROR "Apply the working MD timing update before building Trigger Scintillator replay")
endif()
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include ${MD_PULSE_INCLUDE_DIR} ${PODD_INCLUDE_DIR})
ROOT_GENERATE_DICTIONARY(G__mollertrigscint
 ${CMAKE_CURRENT_SOURCE_DIR}/include/MOLLERTriggerScintillatorPodd.h
 ${CMAKE_CURRENT_SOURCE_DIR}/include/MOLLERTriggerScintillatorDetector.h
 ${CMAKE_CURRENT_SOURCE_DIR}/include/MOLLERTriggerScintillatorApparatus.h
 MODULE mollertrigscint LINKDEF ${CMAKE_CURRENT_SOURCE_DIR}/include/TriggerScintillatorLinkDef.h)
add_library(mollertrigscint SHARED src/MOLLERTriggerScintillatorPodd.cxx src/MOLLERTriggerScintillatorDetector.cxx src/MOLLERTriggerScintillatorApparatus.cxx
 ${CMAKE_CURRENT_BINARY_DIR}/G__mollertrigscint.cxx)
target_include_directories(mollertrigscint PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
 $<INSTALL_INTERFACE:include> ${MD_PULSE_INCLUDE_DIR} ${PODD_INCLUDE_DIR})
target_link_libraries(mollertrigscint PUBLIC ${PODD_LIBRARIES} ROOT::Core ROOT::RIO ROOT::Tree ROOT::Hist ROOT::Physics)
get_filename_component(PODD_LIBRARY_DIR ${PODD_LIBRARY} DIRECTORY)
set_target_properties(mollertrigscint PROPERTIES INSTALL_RPATH "${PODD_LIBRARY_DIR};${ROOT_LIBRARY_DIR}")
install(TARGETS mollertrigscint LIBRARY DESTINATION lib64)
install(DIRECTORY include/ DESTINATION include FILES_MATCHING PATTERN "*.h")
install(FILES "${MD_PULSE_INCLUDE_DIR}/MDPulse.h" DESTINATION include)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/libmollertrigscint_rdict.pcm
 ${CMAKE_CURRENT_BINARY_DIR}/libmollertrigscint.rootmap DESTINATION lib64 OPTIONAL)
message(STATUS "Podd include: ${PODD_INCLUDE_DIR}")
message(STATUS "Podd libraries: ${PODD_LIBRARIES}")

if(BUILD_TESTING)
 add_executable(test_ts_reader tests/test_reader.cpp)
 target_include_directories(test_ts_reader PRIVATE include)
 target_link_libraries(test_ts_reader PRIVATE ROOT::Core ROOT::RIO ROOT::Tree)
 add_test(NAME ts_reader COMMAND test_ts_reader "${CMAKE_CURRENT_SOURCE_DIR}/../../MOLLER-replay/DB_trigscint/MOLLER_trigscint_electronics_map.csv" "${CMAKE_CURRENT_BINARY_DIR}/ts_reader_fixture.root")
endif()
