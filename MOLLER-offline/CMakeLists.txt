cmake_minimum_required(VERSION 3.16)
project(MOLLEROffline VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(BUILD_REPLAY "Build the ROOT/Podd MOLLER replay library" ON)
if(BUILD_TESTING)
 add_executable(test_md_pulse tests/test_pulse.cpp)
 target_include_directories(test_md_pulse PRIVATE include)
 add_test(NAME md_pulse COMMAND test_md_pulse)
endif()
if(NOT BUILD_REPLAY)
 return()
endif()
find_package(ROOT REQUIRED COMPONENTS Core RIO Tree Hist Physics)
include(${ROOT_USE_FILE})
set(PODD_PREFIX "$ENV{ANALYZER}" CACHE PATH "Hall A Analyzer install prefix")
if(NOT PODD_PREFIX)
 find_program(ANALYZER_EXE analyzer)
 if(ANALYZER_EXE)
  get_filename_component(_bin "${ANALYZER_EXE}" DIRECTORY)
  get_filename_component(PODD_PREFIX "${_bin}" DIRECTORY)
 endif()
endif()
find_path(PODD_INCLUDE_DIR THaAnalyzer.h HINTS "${PODD_PREFIX}/include")
find_library(PODD_LIBRARY NAMES HallA Podd HINTS "${PODD_PREFIX}/lib64" "${PODD_PREFIX}/lib")
if(NOT PODD_INCLUDE_DIR OR NOT PODD_LIBRARY)
 message(FATAL_ERROR "Podd not found. Set -DPODD_PREFIX=/path/to/analyzer/install")
endif()
set(PODD_LIBRARIES ${PODD_LIBRARY})
foreach(_lib Podd dc)
 find_library(PODD_${_lib}_LIBRARY NAMES ${_lib} HINTS "${PODD_PREFIX}/lib64" "${PODD_PREFIX}/lib")
 if(PODD_${_lib}_LIBRARY)
  list(APPEND PODD_LIBRARIES ${PODD_${_lib}_LIBRARY})
 endif()
endforeach()
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include ${PODD_INCLUDE_DIR})
ROOT_GENERATE_DICTIONARY(G__moller
 ${CMAKE_CURRENT_SOURCE_DIR}/include/MOLLERMDPodd.h
 ${CMAKE_CURRENT_SOURCE_DIR}/include/MOLLERMDDetector.h
 ${CMAKE_CURRENT_SOURCE_DIR}/include/MOLLERApparatus.h
 MODULE moller LINKDEF ${CMAKE_CURRENT_SOURCE_DIR}/include/MOLLERLinkDef.h)
add_library(moller SHARED src/MOLLERMDPodd.cxx src/MOLLERMDDetector.cxx src/MOLLERApparatus.cxx
 ${CMAKE_CURRENT_BINARY_DIR}/G__moller.cxx)
target_include_directories(moller PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
 $<INSTALL_INTERFACE:include> ${PODD_INCLUDE_DIR})
target_link_libraries(moller PUBLIC ${PODD_LIBRARIES} ROOT::Core ROOT::RIO ROOT::Tree ROOT::Hist ROOT::Physics)
get_filename_component(PODD_LIBRARY_DIR ${PODD_LIBRARY} DIRECTORY)
set_target_properties(moller PROPERTIES INSTALL_RPATH "${PODD_LIBRARY_DIR};${ROOT_LIBRARY_DIR}")
install(TARGETS moller LIBRARY DESTINATION lib64)
install(DIRECTORY include/ DESTINATION include FILES_MATCHING PATTERN "*.h")
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/libmoller_rdict.pcm
 ${CMAKE_CURRENT_BINARY_DIR}/libmoller.rootmap DESTINATION lib64 OPTIONAL)
message(STATUS "Podd include: ${PODD_INCLUDE_DIR}")
message(STATUS "Podd libraries: ${PODD_LIBRARIES}")
